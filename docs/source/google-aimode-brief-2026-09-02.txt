python
import re
import json

def parse_and_normalize_ocr(raw_ocr_text):
    """
    Cleans structural alignment text shifts, removes spacing errors,
    and isolates target financial and wallet tokens from scanned images.
    """
    parsed_data = {}
    
    # 1. Normalize structural noise (e.g., '1 2 3   S a n d b o x' from Page 4)
    normalized_text = re.sub(r'(?<=\b\w)\s(?=\w\b)', '', raw_ocr_text)
    # Collapse multiple tracking white spaces
    normalized_text = " ".join(normalized_text.split())
    
    # 2. Extract FHA Case Number pattern
    case_match = re.search(r'FHA\s?Case\s?(?:No\.?|Number:?)\s?([\d-]+)', normalized_text, re.IGNORECASE)
    if case_match:
        parsed_data['fha_case_number'] = case_match.group(1)
        
    # 3. Extract Cash to Close currency value
    cash_match = re.search(r'Cash\s?to\s?Close[\s\w]*\$\s?([\d,]+\.\d{2})', normalized_text, re.IGNORECASE)
    if cash_match:
        parsed_data['cash_to_close'] = float(cash_match.group(1).replace(',', ''))
        
    # 4. Parse XRPL Ledger Addresses (Base58 check format starting with 'r')
    wallet_matches = re.findall(r'\br[1-9A-HJ-NP-Za-km-z]{25,34}\b', raw_ocr_text)
    if wallet_matches:
        parsed_data['detected_xrpl_wallets'] = list(set(wallet_matches))
        
    return parsed_data

import re
import json

def extract_financial_matrices(raw_ocr_stream):
    """
    Isolates true one-time fees from monthly recurring components
    scanned from the mortgage/escrow bundle.
    """
    extracted = {}
    
    # 1. Clean line breaks and white spaces
    clean_text = " ".join(raw_ocr_stream.split())
    
    # 2. Extract Cash to Close (One-time closing target payload)
    cash_match = re.search(r'Cash\s?to\s?Close[\s\w:]*\$\s?([\d,]+\.\d{2})', clean_text, re.IGNORECASE)
    if cash_match:
        extracted['cash_to_close'] = float(cash_match.group(1).replace(',', ''))
        
    # 3. Extract True Monthly Recurring Payment (Total PITI)
    piti_match = re.search(r'Projected\s?Recurring\s?Payment[\s\w\(\)]*\$\s?([\d,]+\.\d{2})', clean_text, re.IGNORECASE)
    if piti_match:
        extracted['monthly_piti'] = float(piti_match.group(1).replace(',', ''))
        
    # 4. Extract FHA Case Identifier
    case_match = re.search(r'(?:FHA\s?Case|Identifier)\s?(?:No\.?|Number|Key)?\s?([\d-]+)', clean_text, re.IGNORECASE)
    if case_match:
        extracted['fha_case_id'] = case_match.group(1)
        
    return extracted
